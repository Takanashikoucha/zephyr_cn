L2CAP

ECHO

*********************

L2CAP

应用

该

   void

   net_buf
   *buf)
   {
       /*
       Handle
       ECHO
       request
       */

   static
   void
   echo_rsp_cb(struct
   bt_conn
   *conn,
   struct
   net_buf
   *buf)
   {
       /*
       Handle
       ECHO
       response
       */

   static
   struct
   bt_l2cap_br_echo_cb
   echo_cb
   =
   {
       .req
       =
       echo_req_cb,
       .rsp
       =
       echo_rsp_cb,

   bt_l2cap_br_echo_cb_register(&echo_cb);
